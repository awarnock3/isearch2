67// One search hit: the source document's key/doctype/path/file name,
68// its [RecordStart,RecordEnd] byte span within that file, a relevance
69// Score, and (for virtual databases) which DbNum/MDT it came from. Not
70// self-sufficient for reading the record back -- GetRecordData() reads
71// PathName+FileName directly off disk using RecordStart/RecordEnd, not
72// through MyMdt.
73class RESULT {
74public:
75 RESULT();
76 // Deep-copies every field, including MyMdt; safe under
77 // self-assignment.
78 RESULT& operator=(const RESULT& OtherResult);
79 void SetKey(const STRING& NewKey);
80 // Key prefixed with "DbNum:" when DbNum > 0 (virtual-database
81 // results), or just Key otherwise.
82 void GetVKey(STRING* StringBuffer) const;
83 void GetKey(STRING* StringBuffer) const;
84 void SetDocumentType(const STRING& NewDocumentType);
85 void GetDocumentType(STRING* StringBuffer) const;
86 void SetPathName(const STRING& NewPathName);
87 void GetPathName(STRING* StringBuffer) const;
88 void SetFileName(const STRING& NewFileName);
89 void GetFileName(STRING* StringBuffer) const;
90 // PathName immediately followed by FileName, concatenated.
91 void GetFullFileName(STRING* StringBuffer) const;
92 void SetRecordStart(const GPTYPE NewRecordStart);
93 GPTYPE GetRecordStart() const;
94 void SetRecordEnd(const GPTYPE NewRecordEnd);
95 GPTYPE GetRecordEnd() const;
96 void SetScore(const DOUBLE NewScore);
97 DOUBLE GetScore() const;
98#ifdef DO_HIGHLIGHTING
99 void SetHitTable(const FCT& NewHitTable);
100 void GetHitTable(PFCT HitTableBuffer) const;
101#endif
102 // RecordEnd - RecordStart + 1.
103 LONG GetRecordSize() const;
104 // Reads [RecordStart,RecordEnd] out of GetFullFileName() from disk.
105 void GetRecordData(STRING* StringBuffer) const;
106 // Only implemented when DO_HIGHLIGHTING is defined (it isn't
107 // anywhere in this build); a no-op otherwise.
108 void GetHighlightedRecord(const STRING& BeforeTerm,
109 const STRING& AfterTerm,
110 STRING* Buffer) const;
111 void SetDbNum(const INT NewDbNum); // Added for virtual databases
112 INT GetDbNum() const; // Added for virtual databases
113 // Saves a reference to NewMdt (not a copy or an owned pointer).
114 void SetMdt(MDT& NewMdt);
115 MDT* GetMdt() const;
116 ~RESULT();
117private:
118 INT DbNum; // Added for virtual databases
119 STRING Key;
120 STRING DocumentType;
121 STRING PathName;
122 STRING FileName;
123 GPTYPE RecordStart;
124 GPTYPE RecordEnd;
125 DOUBLE Score;
126 MDT* MyMdt; // Save the MDT pointer so we can retrieve the headline
127 // from the right virtual database
128#ifdef DO_HIGHLIGHTING
129 PFCT HitTable;
130#endif
131};
