65typedef struct marc_leader_over /* overlay for marc leader */
66 { char LRECL[5]; /* Logical Record Length */

77 /* Entry Map - description of directory fields */

83typedef struct marc_direntry_over /* overlay for directory entries */
84 { char tag[3]; /* field tag */
85 char flen[4]; /* field length */
86 char fstart[5]; /* field start position*/
87 } MARC_DIRENTRY_OVER;

89typedef struct marc_subfield /* processing structure for subfields */
90 { char *data; /* pointer to data */
91 struct marc_subfield *next; /* pointer to next subfield */
92 char code; /* subfield code char */
93 } MARC_SUBFIELD;
94
95typedef struct marc_field /* field linked list node */
96 { char *data; /* pointer to start of field data */
97 MARC_SUBFIELD *subfield; /* head of linked list of subfields */
98 MARC_SUBFIELD *lastsub; /* tail of linked list of subfields */
99 struct marc_field *next; /* next field in list */
100 char tag[4]; /* 3 char tag + null */
101 char indicator1; /* first indicator */
102 char indicator2; /* second indicator */
103 char subfcodes[21]; /* string of subfield codes */
104 /* element [0] is a count of subfields */
105 int length; /* full field length */
106 } MARC_FIELD;
107
108typedef struct marc_rec /* marc record processing format */
109 { INT4 length;
110 char *record,
111 *BaseAddr;
112 MARC_LEADER_OVER *leader;
113 char *fixed_fields;
114 MARC_FIELD *fields; /* head of linked list */
115 MARC_FIELD *lastfield; /* tail of linked list */
116 int nfields;
117 } MARC_REC;
