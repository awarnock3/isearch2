58// File Pointer Table: a small LRU cache of open FILE* handles, so
59// repeatedly ffopen()-ing the same file reuses an already-open handle
60// instead of paying a real open() every time. ffclose() only marks an
61// entry as logically closed (available for reuse); the physical
62// fclose() is deferred to CloseAll(), an LRU eviction, or a genuine
63// open-mode change -- see ffopen()'s cache-hit branches.
64class FPT {
65public:
66 FPT();
67 FPT(const INT TableSize);
68 // BUGFIX #1: FPT owns a heap-allocated FPREC* Table array, where each
69 // entry caches a live FILE*, but declared no copy constructor and no
70 // operator= at all -- not even a hand-written one -- so the
71 // compiler-generated ones did a member-wise shallow copy. Confirmed
72 // to double-free Table under ASan (surfacing inside CloseAll() during
73 // destruction). Made explicitly non-copyable rather than deep-copied:
74 // copying a table of open file handles has no single obviously-
75 // correct meaning (duplicate the descriptor? reopen by name? leave
76 // the copy closed?), the same reasoning as MDT's non-copyable choice
77 // this batch. See docs/BUG_CATALOG.md.
78 FPT(const FPT&) = delete;
79 FPT& operator=(const FPT&) = delete;
80 PFILE ffopen(const STRING& FileName, const CHR *Type);
81 INT ffclose(FILE *FilePointer);
82 void CloseAll();
83 ~FPT();
84
85private:
86 void Init(const INT TableSize);
87 INT Lookup(const STRING& FileName);
88 INT Lookup(const FILE *FilePointer);
89 void HighPriority(const INT Index);
90 void LowPriority(const INT Index);
91 FPREC *Table;
92 INT TotalEntries;
93 INT MaximumEntries;
94};
