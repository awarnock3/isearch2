65// A Multiple Document Table record: one indexed document's key,
66// doctype, path/file name, and its [GlobalFileStart,GlobalFileEnd) /
67// [LocalRecordStart,LocalRecordEnd) byte-offset spans within the
68// corpus. All string fields are fixed-size CHR buffers (not STRING),
69// matching this record's on-disk, fixed-length layout (see
70// MDT::GetEntry/AddEntry, src/mdt.cxx, which fread()/fwrite() it as a
71// raw block) -- Set* accessors always null-terminate within bounds
72// (via STRING::GetCString); Get* accessors never assume the buffer
73// already is, since a corrupt or truncated on-disk record can leave
74// one that isn't (see BUGFIX #1/#2 in source).
75class MDTREC {
76public:
77 MDTREC();
78 // Deep-copies every field; safe under self-assignment. Bounded even
79 // if OtherMdtRec's buffers aren't null-terminated.
80 MDTREC& operator=(const MDTREC& OtherMdtRec);
81 void SetKey(const STRING& NewKey);
82 void GetKey(STRING* StringBuffer) const;
83 void SetDocumentType(const STRING& NewDocumentType);
84 void GetDocumentType(STRING* StringBuffer) const;
85 void SetPathName(const STRING& NewPathName);
86 void GetPathName(STRING* StringBuffer) const;
87 void SetFileName(const STRING& NewFileName);
88 void GetFileName(STRING* StringBuffer) const;
89 // PathName immediately followed by FileName, concatenated.
90 void GetFullFileName(STRING* StringBuffer) const;
91 void SetGlobalFileStart(const GPTYPE NewGlobalFileStart);
92 GPTYPE GetGlobalFileStart() const;
93 void SetGlobalFileEnd(const GPTYPE NewGlobalFileEnd);
94 GPTYPE GetGlobalFileEnd() const;
95 void SetLocalRecordStart(const GPTYPE NewLocalRecordStart);
96 GPTYPE GetLocalRecordStart() const;
97 void SetLocalRecordEnd(const GPTYPE NewLocalRecordEnd);
98 GPTYPE GetLocalRecordEnd() const;
99 void SetDeleted(const GDT_BOOLEAN Flag);
100 GDT_BOOLEAN GetDeleted() const;
101 // Byte-swaps the four GPTYPE offset fields in place, for cross-endian
102 // file I/O.
103 void FlipBytes();
104 ~MDTREC();
105
106private:
107 CHR Key[DocumentKeySize];
108 CHR DocumentType[DocumentTypeSize];
109 CHR PathName[DocPathNameSize];
110 CHR FileName[DocFileNameSize];
111 GPTYPE GlobalFileStart;
112 GPTYPE GlobalFileEnd;
113 GPTYPE LocalRecordStart;
114 GPTYPE LocalRecordEnd;
115 CHR Deleted;
116};
