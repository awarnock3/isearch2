58// A tree of named nodes -- first-child/next-sibling representation
59// (Next chains a node's siblings, Child points to its first child) --
60// used to hold structured config/profile data (e.g. parsed INI-style
61// "[section]\nkey=val1,val2" files via Profile*, or the nested
62// "<tag>...</tag>" meta-defaults format via parseMetaDefaults() below)
63// addressable by a STRLIST path of node names down from the root.
64class REGISTRY {
65public:
66 REGISTRY(const STRING& Title);
67 REGISTRY(const CHR *Title);
68 // Deep-copies OtherRegistry's whole subtree (via clone() below);
69 // safe under self-assignment.
70 REGISTRY& operator=(const REGISTRY& OtherRegistry);
71 // Recursively deep-copies this node and its whole subtree.
72 REGISTRY* clone() const;
73 void PrintSgml(FILE* fp, const STRLIST& Position);
74 // Replaces Position's children with Value's entries (one child node
75 // per entry).
76 void SetData(const STRLIST& Position, const STRLIST& Value);
77 // Appends Value's entries as additional children of Position.
78 void AddData(const STRLIST& Position, const STRLIST& Value);
79 // Fills *StrlistBuffer with Position's direct children's data, or
80 // clears it if Position doesn't exist.
81 void GetData(const STRLIST& Position, STRLIST *StrlistBuffer);
82 void SaveToFile(const STRING& FileName, const STRLIST& Position);
83 void LoadFromFile(const STRING& FileName, const STRLIST& Position);
84 void AddFromFile(const STRING& FileName, const STRLIST& Position);
85 // Note: ProfileGetString() and ProfileWriteString() consider
86 // a comma-delimited string in Entry to be a list of values, and
87 // will store each as a separate node in the registry. That is
88 // the deepest level of nesting that these two functions allow.
89 // For access to full nesting, I recommend using SetData()/AddData()
90 // and GetData().
91 void ProfileGetString(const STRING& Section, const STRING& Entry,
92 const STRING& Default, STRING *StringBuffer);
93 void ProfileWriteString(const STRING& Section, const STRING& Entry,
94 const STRING& StringData);
95 // Note: ProfileLoadFromFile() and ProfileAddFromFile() also parse
96 // a comma-delimited list (e.g., tag=val1,val2,val3) into multiple nodes.
97 void ProfileLoadFromFile(const STRING& FileName, const STRLIST& Position);
98 void ProfileAddFromFile(const STRING& FileName, const STRLIST& Position);
99 void ProfileWrite(ostream& os, const STRING& FileName, const STRLIST& Position);
100// friend ostream & operator<<(ostream& os, const REGISTRY& Registry);
101 ~REGISTRY();
102
103private:
104 void PrintSgml(FILE* fp);
105 void ProfileAddFromFile(const STRING& FileName);
106 void LoadFromFile(const STRING& FileName);
107 void AddFromFile(const STRING& FileName);
108// void Print(ostream& os, const INT Level) const;
109 void fprint(FILE* fp, const INT level = 0) const;
110 void ProfilePrint(ostream& os, const INT Level) const;
111 void GetData(STRLIST *StrlistBuffer);
112 void DeleteChildren();
113 REGISTRY* FindNode(const STRING& Position);
114 REGISTRY* GetNode(const STRING& Position);
115 REGISTRY* FindNode(const STRLIST& Position);
116 REGISTRY* GetNode(const STRLIST& Position);
117 void SetData(const STRLIST& Value);
118 void AddData(const STRLIST& Value);
119 STRING Data;
120 REGISTRY* Next;
121 REGISTRY* Child;
122};
