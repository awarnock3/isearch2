56// Free-function grab bag used tree-wide: filesystem path helpers
57// (AddTrailingSlash/RemovePath/RemoveFileName/RemoveFileExtension/
58// ExpandFileSpec), file/db existence and size checks (GetFileSize/
59// IsFile/DBExists), endianness helpers (GpSwab/IsBigEndian), and a
60// handful of string/date utilities (IsAlnum/trimWhitespace/
61// ParseIsoDate). No class here, no state -- every function is
62// independently callable given just its arguments.

101// Appends DIR_SLASH ('/' on UNIX) unless PathName is empty or already
102// ends in one.
103void AddTrailingSlash(STRING* PathName);
104// Keeps only the last path component: erases everything up to and
105// including the last DIR_SLASH. A no-op if there's no DIR_SLASH.
106void RemovePath(STRING* FileName);
107// Keeps only the directory prefix (through the last DIR_SLASH,
108// inclusive): the complement of RemovePath(). No DIR_SLASH means no
109// directory, so the result is the empty string, not the input
110// unchanged -- see RemovePath() for the case that IS a no-op.
111void RemoveFileName(STRING* PathName);
112// Erases everything after the last '.' (the dot itself is kept). A
113// no-op if PathName has no '.'.
114void RemoveFileExtension(STRING* PathName);
