57// One entry in FPT's (src/fpt.hxx) open-file table: a file name, the
58// FILE* it's currently open under (if any -- FPREC doesn't open/close
59// it itself, just records FPT's bookkeeping), that file's open mode,
60// and an LRU-style Priority plus a Closed flag FPT uses to decide which
61// file to close when the table is full.
62class FPREC {
63public:
64 FPREC();
65 // Copies every field, including Priority and Closed; safe under
66 // self-assignment.
67 FPREC& operator=(const FPREC& OtherFprec);
68 // Stores NewFileName expanded to an absolute path (see
69 // common.hxx's ExpandFileSpec).
70 void SetFileName(const STRING& NewFileName);
71 void GetFileName(STRING *StringBuffer) const;
72 void SetFilePointer(FILE* NewFilePointer);
73 PFILE GetFilePointer() const;
74 void SetPriority(const INT NewPriority);
75 INT GetPriority() const;
76 void SetClosed(const GDT_BOOLEAN NewClosed);
77 GDT_BOOLEAN GetClosed() const;
78 void SetOpenMode(const STRING& NewOpenMode);
79 void GetOpenMode(STRING *StringBuffer) const;
80 ~FPREC();
81
82private:
83 STRING FileName;
84 PFILE FilePointer;
85 STRING OpenMode;
86 INT Priority;
87 GDT_BOOLEAN Closed;
88};
