76// The internal, working search result set the query engine builds up
77// while evaluating a query (as opposed to RSET, the materialized,
78// user-facing result set produced at the end via GetRset()/Fill()).
79class IRSET : public OPERAND {
80public:
81 IRSET(const PIDBOBJ DbParent);
82 // BUGFIX #2: added -- previously absent, so copy-constructing an
83 // IRSET (e.g. `IRSET b = a;`, pass/return by value) used the
84 // compiler-generated shallow copy of Table, and both copies'
85 // destructors then deleted the same heap array: confirmed
86 // double-free/use-after-free under ASan. Deep-copies entries the
87 // same way operator= does, below.
88 IRSET(const IRSET& OtherIrset);

95 // same MdtIndex (accumulating hit count/score) if one exists. O(n).
96 void AddEntry(const IRESULT& ResultRecord, const INT AddHitCounts);
97 // Like AddEntry, but always appends without checking for an existing
98 // entry with the same MdtIndex -- callers are responsible for later
99 // sorting (SortByIndex) and deduplicating (MergeEntries) themselves.
100 void FastAddEntry(const IRESULT& ResultRecord, const INT AddHitCounts);

102 // must already be sorted by index (SortByIndex) for this to work.
