53// A singly linked stack of owned, heap-allocated OPOBJ* entries (query
54// operators/operands), used while evaluating a query's RPN expression.
55// operator<<(OPOBJ&) pushes a Duplicate() of its argument; operator<<
56// (OPOBJ*) takes ownership of the pointer directly. Either way, popped
57// entries (via operator>>) become the caller's responsibility to delete.
58class OPSTACK {
59public:
60 OPSTACK();
61 // BUGFIX #2: added -- previously absent, so copy-constructing an
62 // OPSTACK (e.g. `OPSTACK b = a;`) used the compiler-generated shallow
63 // copy of Head. This was harmless only because ~OPSTACK() didn't free
64 // anything (see BUGFIX #3); now that it does, an unguarded shallow
65 // copy would double-free the shared chain of nodes the same way
66 // DFT/RSET/IRSET's missing copy constructors did. Deep-copies
67 // entries the same way operator= does, below.
68 OPSTACK(const OPSTACK& OtherOpstack);
69 OPSTACK& operator=(const OPSTACK& OtherOpstack);
70 OPSTACK& operator<<(OPOBJ& Op);
71 OPSTACK& operator<<(OPOBJ* Op);
72 POPOBJ operator>>(POPOBJ& OpPtr);
73 PIRSET operator>>(PIRSET& OpPtr);
74 void Reverse();
75 ~OPSTACK();
76private:
77 void Push(OPOBJ& Op);
78 void Push(OPOBJ *Op);
79 POPOBJ Pop();
80 POPOBJ Head;
81};
