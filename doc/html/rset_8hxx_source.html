52// A dynamically resizing array of RESULT entries (a materialized,
53// user-facing search result set), sortable by key or score.
54class RSET {
55public:
56 RSET();
57 // BUGFIX #2: added -- previously absent, so copy-constructing an RSET
58 // (e.g. `RSET b = a;`, pass/return by value) used the compiler-
59 // generated shallow copy of Table, and both copies' destructors then
60 // deleted the same heap array: confirmed double-free/use-after-free
61 // under ASan. Deep-copies entries the same way operator= does, below.
62 RSET(const RSET& OtherRset);
63 // BUGFIX #2: added for the same reason -- previously absent entirely
64 // (not even a shallow one), so `b = a;` used the compiler-generated
65 // shallow copy-assignment, confirmed to double-free the same way.
66 RSET& operator=(const RSET& OtherRset);
67 // Replaces this table's contents with FileName's contents, previously
68 // written by SaveTable(). See BUGFIX #4 for why this matters: this
69 // isn't a raw memory dump of Table (RESULT's STRING fields own heap
70 // buffers that can't survive one), but a field-by-field serialization
71 // through RESULT's own public getters/setters.
72 void LoadTable(const STRING& FileName);
73 void SaveTable(const STRING& FileName);
74 void AddEntry(const RESULT& ResultRecord);
75 // Copies the Index'th entry (1-based) into *ResultRecord; leaves
76 // *ResultRecord untouched if Index is out of [1, GetTotalEntries()]
77 // range.
78 void GetEntry(const INT Index, PRESULT ResultRecord) const;
79 // Sorts entries ascending by key.
80 void SortByKey();
81 // Sorts entries descending by score (best match first), matching
82 // IRSET::SortByScore's convention for the same conceptual operation.
83 void SortByScore();
84 INT GetScaledScore(const DOUBLE UnscaledScore, const INT ScaleFactor);
85 void Expand();
86 void CleanUp();
87 void Resize(const SIZE_T Entries);
88 // 1-based, like GetEntry.
89 void SetEntry(const INT x, const RESULT& ResultRecord );
90 SIZE_T GetTotalEntries();
91 void SetScoreRange(DOUBLE High, DOUBLE Low);
92// void Dump() const;
93 ~RSET();
94private:
95 PRESULT Table;
96 SIZE_T TotalEntries;
97 SIZE_T MaxEntries;
98 DOUBLE HighScore, LowScore;
99};
