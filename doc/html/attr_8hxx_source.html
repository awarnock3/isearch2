54// A single Z39.50/GILS-style search attribute: an attribute-set id, an
55// attribute type (Use/Relation/Position/Structure/Truncation/
56// Completeness, per the protocol), and its value -- stored as STRING
57// regardless of which SetAttrValue overload set it, so GetAttrValue()
58// (INT) parses it back out via STRING::GetInt().
59class ATTR {
60public:
61 ATTR();
62 ATTR& operator=(const ATTR& OtherAttr);
63 void SetSetId(const STRING& NewSetId);
64 void GetSetId(PSTRING StringBuffer) const;
65 void SetAttrType(const INT NewAttrType);
66 INT GetAttrType() const;
67 void SetAttrValue(const STRING& NewAttrValue);
68 void GetAttrValue(PSTRING StringBuffer) const;
69 void SetAttrValue(const INT NewAttrValue);
70 // Parses the stored value as an integer (0 if it isn't one).
71 INT GetAttrValue() const;
72 ~ATTR();
73private:
74 STRING SetId;
75 INT AttrType;
76 STRING AttrValue;
77};
