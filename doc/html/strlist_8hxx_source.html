58// An ordered list of STRING entries, one per node: each STRLIST object
59// is itself a VLIST node holding a single entry's String, and the
60// list's identity is the head node's `this` (see VLIST for the
61// circular-list mechanics AddEntry/GetEntry/etc. build on).
62//
63// NOTE: STRLIST has no explicit copy constructor, only operator=
64// (below). VLIST's own copy constructor is compiler-generated and
65// shallow-copies its Next/Prev pointers instead of splicing into the
66// circle -- a known, tracked defect (see
67// docs/AUTOPILOT_LOG.md#srcvlisthxx, blocked pending a header-signature
68// decision on vlist.hxx). STRLIST inherits that same latent risk for
69// copy-CONSTRUCTION (as opposed to assignment, which this file's
70// operator= handles correctly); every call site in this tree already
71// avoids it by default-constructing and then assigning, as Split()
72// does below. Fixing it requires adding a declaration to vlist.hxx, so
73// it's deferred there rather than duplicated as a second blocked row
74// here.
75class STRLIST
76 : public VLIST
77{
78
79public:
80 STRLIST();
81 // Deep-copies OtherStrlist's entries via new nodes (never copies
82 // Next/Prev pointers directly) -- safe even for self-assignment.
83 STRLIST& operator=(const STRLIST& OtherStrlist);
84 void AddEntry(const STRING& StringEntry);
85 void AddEntry(const CHR* Entry);
86 // 1-based. Beyond the current entry count, appends empty filler
87 // entries up to Index. A no-op for Index 0.
88 void SetEntry(const SIZE_T Index, const STRING& StringEntry);
89 // 1-based. Leaves *StringEntry unchanged for an out-of-range Index.
90 void GetEntry(const SIZE_T Index, STRING* StringEntry) const;
91 // Splits TheString on every occurrence of Separator into entries,
92 // replacing this list's current contents. A trailing empty segment
93 // (Separator at the very end) is dropped, not added as an entry.
94 void Split(const CHR* Separator, const STRING& TheString);
95 void Split(const CHR Separator, const STRING& TheString);
96 // Concatenates all entries, interleaving Separator between them.
97 void Join(const CHR* Separator, STRING* StringBuffer);
98 // Case-insensitive linear search; returns the 1-based index of the
99 // first match, or 0 if none.
100 SIZE_T SearchCase(const STRING& SearchTerm);
101 // Treats each entry as a "Title = value" pair (whitespace around
102 // Title is trimmed before the case-insensitive compare) and returns
103 // the value for the first matching Title via *StringBuffer, or ""
104 // if none match.
105 void GetValue(const CHR* Title, STRING* StringBuffer);
106 void GetValue(const STRING& Title, STRING* StringBuffer);
107 // Writes each entry to fp, one per line.
108 void Dump(const PFILE fp) ;
109// friend ostream & operator<<(ostream& os, const STRLIST& str);
110
111private:
112 STRING String;
113};
