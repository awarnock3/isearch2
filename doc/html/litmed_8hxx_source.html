29// Unlike doctype/html.hxx's equivalent STRICT_HTML (guarded with
30// #ifndef so it can be overridden by a compiler flag), this is an
31// unconditional #define -- any -DSTRICT_LITMED=0 on the command line
32// gets silently redefined back to 1 by this line, so LITMED::
33// ParseFields()'s #else branch (IgnoreLITMEDTag()) is permanently
34// unreachable in practice, same as doctype/html.cxx's own genuinely-
35// dead `#if 1` branch. Left as its own historical curiosity rather
36// than "fixed" into an #ifndef, since making the #else branch
37// actually reachable would be a behavior change with no clear
38// specification for what the #else path should accept.
39#define STRICT_LITMED 1
