16// Data Field Table: a dynamically resizing array of DF (Data Field) entries,
17// one per named field found while parsing a document, 1-indexed to match
18// GetEntry/AddEntry's own convention below.
19class DFT {
20public:
21 DFT();
22 // BUGFIX #2: added -- previously absent, so copy-constructing a DFT
23 // (e.g. `DFT b = a;`, pass/return by value) used the compiler-generated
24 // copy constructor, which shallow-copies the Table pointer instead of
25 // the entries it points to. Both copies' destructors then delete the
26 // same heap array, corrupting memory on the first destruction and
27 // double-freeing on the second. Deep-copies entries the same way
28 // operator= does, below.
29 DFT(const DFT& OtherDft);
30 // Allocates a fresh, empty 100-entry Table. Used internally by the
31 // constructors; calling it directly on a DFT that already owns entries
32 // overwrites Table without freeing it first and leaks the old array,
33 // so prefer assignment (operator=) or a fresh DFT instead.
34 void Init();
35 DFT& operator=(const DFT& OtherDft);
36 // Appends a copy of DfRecord, expanding Table first if it's full.
37 void AddEntry(const DF& DfRecord);
38 // Copies the Index'th entry (1-based) into *DfRecord; leaves *DfRecord
39 // untouched if Index is out of [1, GetTotalEntries()] range.
40 void GetEntry(const INT Index, PDF DfRecord) const;
41 // Grows Table's capacity by 100 entries.
42 void Expand();
43 // Shrinks Table's capacity down to exactly GetTotalEntries().
44 void CleanUp();
45 // Reallocates Table to hold exactly Entries entries, copying over
46 // min(Entries, GetTotalEntries()) existing ones.
47 void Resize(const INT Entries);
48 INT GetTotalEntries() const;
49 // Serializes the table as text to fp.
50 void Write(PFILE fp) const;
51 // Reads the table back from text previously written by Write().
52 void Read(PFILE fp);
53 ~DFT();
54private:
55 PDF Table;
56 INT TotalEntries;
57 INT MaxEntries;
58};
