51// Doubly linked *circular* list base class: every node's Next/Prev
52// eventually loops back to itself. Carries no payload of its own --
53// subclasses (e.g. FCT, STRLIST) add their own data per node.
54class VLIST {
55public:
56 VLIST();
57 // BUGFIX #1: no copy constructor and no working operator= existed
58 // (the commented-out sketch below was pure-virtual and never
59 // compiled). A copied node becomes the sole member of its own new
60 // one-node circle -- VLIST carries no other state, so this is
61 // identical to default-construction. The source's own Next/Prev
62 // links are deliberately never read: splicing a node into someone
63 // else's circle needs AddNode()'s bookkeeping, not a blind pointer
64 // copy. Not virtual, unlike the old sketch -- nothing in this tree
65 // assigns/copy-constructs through a VLIST& or VLIST* today; each
66 // subclass that needs to duplicate a circle's *contents* already
67 // does so at its own level (see e.g. FCT::operator=, which walks
68 // the source circle and builds fresh nodes via AddNode()).
69 VLIST(const VLIST& OtherVlist);
70 VLIST& operator=(const VLIST& OtherVlist);
71 // virtual VLIST& operator=(const VLIST& OtherVlist) = 0;
72 virtual void Clear();
73 // virtual void EraseAfter(const SIZE_T Index);
74 virtual void EraseAfter(const INT Index);
75 virtual void Reverse();
76 // virtual SIZE_T GetTotalEntries() const;
77 virtual INT GetTotalEntries() const;
78 virtual ~VLIST();
79protected:
80 virtual void AddNode(VLIST* NewEntryPtr);
81 // virtual VLIST* GetNodePtr(const SIZE_T Index) const;
82 virtual VLIST* GetNodePtr(const INT Index) const;
83 virtual VLIST* GetNextNodePtr() const;
84private:
85 VLIST* Next;
86 VLIST* Prev;
87};
