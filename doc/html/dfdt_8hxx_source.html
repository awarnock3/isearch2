64// Data Field Definitions Table: an ordered, resizable array of DFD
65// entries, one per indexed field, tracking each field's assigned file
66// number and attributes.
67class DFDT {
68public:
69 DFDT();
70 // BUGFIX #1: DFDT owned a heap-allocated Table array but declared no
71 // copy constructor, so the compiler-generated one did a shallow
72 // pointer copy -- confirmed to double-free Table under ASan.
73 // Deep-copies Table (sized to the source's MaxEntries), TotalEntries,
74 // MaxEntries, and Changed, mirroring operator='s own logic. See
75 // docs/BUG_CATALOG.md.
76 DFDT(const DFDT& OtherDfdt);
77 DFDT& operator=(const DFDT& OtherDfdt);
78 void Initialize();
79 void LoadTable(const STRING& FileName);
80 void SaveTable(const STRING& FileName);
81 void AddEntry(const DFD& DfdRecord);
82 void FastAddEntry(const DFD& DfdRecord);
83 void GetEntry(const INT Index, PDFD DfdRecord) const;
84 // Leaves *DfdRecord untouched if FieldName isn't found -- same
85 // convention as GetEntry() above (see BUGFIX #3 in source: this used
86 // to try to signal "not found" by nulling the DfdRecord parameter
87 // itself, which is a by-value pointer and has no effect the caller
88 // can observe).
89 void GetDfdRecord(const STRING& FieldName, PDFD DfdRecord) const;
90 INT GetNewFileNumber() const;
91 void Expand();
92 void CleanUp();
93 void Resize(const INT Entries);
94 INT GetTotalEntries() const;
95 INT GetChanged() const;
96 ~DFDT();
97
98private:
99 PDFD Table;
100 INT TotalEntries;
101 INT MaxEntries;
102 INT Changed;
103};
