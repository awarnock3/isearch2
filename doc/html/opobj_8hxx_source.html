75// Base class for every node OPSTACK (src/opstack.hxx) can hold: leaf
76// operands (e.g. STERM, IRSET) and operators alike, so a query's
77// infix-to-RPN evaluation can push/pop them polymorphically through
78// one stack. Nearly every method here defaults to a no-op/zero return
79// (deliberately, not an oversight -- overridden only by the subset of
80// derived classes it actually applies to) except the four pure
81// virtuals, which every derived class must implement. Next/SetNext/
82// GetNext are OPSTACK's own private linked-list bookkeeping for the
83// nodes it holds, not part of this class's public interface.
84class OPOBJ {
85friend class OPSTACK;
86public:
87 OPOBJ();
88 virtual INT GetOpType() const = 0;
89 virtual INT GetOperandType() const = 0;
90 virtual OPOBJ* Duplicate() const = 0;
91 virtual OPOBJ& operator=(const OPOBJ& OtherOp) = 0;
92 virtual void SetAttributes(const ATTRLIST& NewAttributes) { };
93 virtual void GetAttributes(PATTRLIST AttributesBuffer) const { };
94 virtual void SetTerm(const STRING& NewTerm) { };
95 virtual void GetTerm(PSTRING StringBuffer) const { };
96 virtual void SetOperatorType(const INT OperatorType) { };
97 virtual INT GetOperatorType() const { return 0; };
98 virtual INT GetTotalEntries() const { return 0; };
99// virtual void AddEntry(const IRESULT& ResultRecord, const INT AddHitCounts) { };
100 virtual void GetEntry(const INT Index, PIRESULT ResultRecord) const { };
101 virtual void Or(const OPOBJ& OtherIrset) { };
102 virtual void And(const OPOBJ& OtherIrset) { };
103 virtual void AndNot(const OPOBJ& OtherIrset) { };
104 virtual void Near(const OPOBJ& OtherIrset) { };
105 virtual void SetParent(PIDBOBJ const NewParent) { };
106 virtual PIDBOBJ GetParent() const { return 0; };
107 virtual ~OPOBJ();
108private:
109 void SetNext(OPOBJ* const OpPtr);
110 OPOBJ* GetNext() const;
111 OPOBJ* Next;
112};
