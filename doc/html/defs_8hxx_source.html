243// cout` -- an unqualified name with nothing in this header to declare
244// it. It only ever compiled because every real caller happened to
245// separately pull in `using namespace std;` (via string.hxx) ahead of
246// this header -- true today, not guaranteed. Qualifying it here and
247// including <iostream> above makes COUT work regardless of what else
248// a caller has included.

251// Found but deliberately NOT changed here (see docs/BUG_CATALOG.md#srcdefshxx,
252// "found but out of scope"): none of these three wrap their body in
253// `do { ... } while(0)`, the standard hygiene fix for multi-statement
254// macros, so a use like `if (x) EXIT_ERROR; else ...` can silently
255// misparse. Left alone because src/result.cxx:229 -- already compiled
256// by `make tests` -- invokes EXIT_ERROR with no trailing semicolon,
257// relying on today's bare-`{}` expansion; wrapping in do/while(0)
258// would break that call site, which is out of scope for this turn.
259#define EXIT_ERROR {fflush(stdout); fflush(stderr); exit(1);}
260#define RETURN_ERROR {fflush(stdout); fflush(stderr); return(1);}
261#define RETURN_ZERO {fflush(stdout); fflush(stderr); return(0);}
262#endif
