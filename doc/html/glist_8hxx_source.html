79// A generic intrusive-cell doubly-linked list of untyped GATOM*
80// pointers (each cell also tags its atom with a LIST_* type code, or a
81// caller-defined one). GLIST owns and heap-allocates the GPOSITION
82// cells themselves (InsertAfter()/InsertBefore() `new` them, Delete()
83// `delete`s them) but never owns the GATOM data they point to -- see
84// Delete()'s precondition. Note: GLIST has no destructor, so any cells
85// still in the list when a GLIST is destroyed are leaked; callers are
86// expected to Delete() every remaining cell first (see
87// docs/BUG_CATALOG.md#srcglisthxx for why this wasn't changed --
88// fixing it needs a header change).
89class GLIST {
90public:
91 GLIST();
92 GDT_BOOLEAN IsEmpty();
93 GPOSITION* First();
94 GPOSITION* Last();
95 GPOSITION* Next(GPOSITION *c);
96 GPOSITION* Prev(GPOSITION *c);
97 // Inserts a new cell holding a (tagged with Type) immediately before
98 // c, by inserting after c and then swapping the two cells' contents
99 // -- c keeps its identity/position for any other GPOSITION* already
100 // pointing at it, but now holds the new data.
101 GDT_BOOLEAN InsertBefore(GPOSITION *c, GATOM *a, int Type);
102 // c == nullptr is only valid when the list is empty (starts a new
103 // Head/Tail); otherwise inserts immediately after c.
104 GDT_BOOLEAN InsertAfter(GPOSITION *c, GATOM *a, int Type);
105 // Same as the 4-argument overloads, tagged with LIST_PTR.
106 GDT_BOOLEAN InsertBefore(GPOSITION *c, GATOM *a);
107 GDT_BOOLEAN InsertAfter(GPOSITION *c, GATOM *a);
108 GATOM* Retrieve(GPOSITION *c);
109 // Replaces c's atom in place; does not change c's type tag.
110 GDT_BOOLEAN Update(GPOSITION *c, GATOM *a);
111 INT GetLength();
112 INT DataType(GPOSITION *c);
113 // Unlinks and frees cell c (not its atom -- see the class comment
114 // above). Any other GPOSITION* pointing at c becomes dangling.
115 void Delete(GPOSITION *c);
116
117private:
118 INT Length; /* Number of cells in list */
119 GPOSITION *Head, *Tail; /* Pointers to head and tail of list */
120};
