58// A search hit as tracked internally during scoring/merging (as
59// opposed to RESULT, src/result.hxx, which is the externally-facing
60// form materialized once a hit is finalized): just an MDT row index,
61// running hit count and score, and -- for virtual databases -- which
62// DbNum/MDT it came from.
63class IRESULT {
64public:
65 IRESULT();

67 IRESULT& operator=(const IRESULT& OtherIresult);
68 void SetMdtIndex(const INT NewMdtIndex);
69 INT GetMdtIndex() const;
70 void SetHitCount(const INT NewHitCount);
71 void IncHitCount();
72 void IncHitCount(const INT AddCount);
73 INT GetHitCount() const;
74 void SetScore(const DOUBLE NewScore);
75 void IncScore(const DOUBLE AddScore);
76 DOUBLE GetScore() const;
77#ifdef DO_HIGHLIGHTING
78 void SetHitTable(const FCT& NewHitTable);
79 void GetHitTable(PFCT HitTableBuffer) const;
80 // Appends ResultRecord's HitTable entries onto this one's.
81 void AddToHitTable(const IRESULT& ResultRecord);
82#endif
83 // Saves a reference to NewMdt (not a copy or an owned pointer).
84 void SetMdt(MDT& NewMdt);
85 MDT* GetMdt() const;
86 void SetDbNum(const INT NewDbNum);
87 INT GetDbNum() const;
88 ~IRESULT();
89private:
90 INT MdtIndex;
91 INT HitCount;
92 DOUBLE Score;
93#ifdef DO_HIGHLIGHTING
94 PFCT HitTable;
95#endif
96 INT DbNum; // What virtual database did the record come from?

98 // from the right virtual database
99
100};
