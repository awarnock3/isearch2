60// [start,end) byte-offset occurrences within a document that belong to
61// that field.

72 // Serializes as text: field name, a newline, then FCT::Write's own
73 // format.
74 void Write(FILE *fp) const;
75 // Reads the pair back from text previously written by Write().
76 void Read(FILE *fp);

82};
