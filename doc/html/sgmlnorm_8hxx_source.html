20// A generic SGML/XML-tag-shaped DOCTYPE: any <tag>value</tag> pair
21// becomes one field named after the (UnifiedName-mapped) tag, and any
22// bare attribute on a start tag (<tag attr="val">) becomes a field
23// named "tag@attr" (see store_attributes()). Base class for the
24// tag-parsing DOCTYPEs (e.g. SGMLTAG, HTML) that share this machinery.

27public:

34 STRING *StringBuffer);

37 // hooks into the guts of the field parser
38 virtual const CHR* UnifiedName (const CHR *tag) const; // for children to play with
39
40 /* SGML helper functions */
41 // Splits b (len bytes) in place into a nullptr-terminated array of
42 // pointers to each tag's contents (the '>' bytes are overwritten
43 // with '\0' to terminate each tag string); returns nullptr if a tag
44 // is left unterminated (no closing '>') or on allocation failure.
45 // Caller owns the returned array (delete[] it; the CHR* elements
46 // point into b, not separately allocated).
47 PCHR *parse_tags (CHR *b, GPTYPE len) const;
48 // Scans the nullptr-terminated tag list t for a "/"+tag closing tag
49 // matching tag's name (up to its first space), case-insensitively.
50 // Returns a pointer to the matching entry in t, or nullptr if none.
51 const CHR* find_end_tag (char *const *t, const char *tag) const;
52 // Parses a start tag's "attr=val ..." content (base_ptr is the
53 // record buffer parse_tags() sliced tag_ptr from, needed to compute
54 // field-coordinate offsets) into one field per attribute, named
55 // "tagname@attrname".
56 void store_attributes (PDFT pdft, CHR *base_ptr, CHR *tag_ptr) const;
57
58};
